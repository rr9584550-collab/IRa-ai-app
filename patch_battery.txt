    private var isTtsInitialized = false
    private var pendingCallAnnouncement: String? = null
    
    private var lastBatteryLevel = -1
    private val batteryReceiver = object : BroadcastReceiver() {
        override fun onReceive(context: Context, intent: Intent) {
            when (intent.action) {
                Intent.ACTION_POWER_CONNECTED -> {
                    if (isTtsInitialized) {
                        tts?.speak("ira charging connect janu", TextToSpeech.QUEUE_FLUSH, null, "ChargeConnected")
                    }
                }
                Intent.ACTION_BATTERY_CHANGED -> {
                    val level = intent.getIntExtra(BatteryManager.EXTRA_LEVEL, -1)
                    val scale = intent.getIntExtra(BatteryManager.EXTRA_SCALE, -1)
                    val status = intent.getIntExtra(BatteryManager.EXTRA_STATUS, -1)
                    
                    if (level != -1 && scale != -1) {
                        val batteryPct = level * 100 / scale.toFloat()
                        if (batteryPct >= 100f && status == BatteryManager.BATTERY_STATUS_FULL && lastBatteryLevel != level) {
                            if (isTtsInitialized) {
                                tts?.speak("charge full hogaya he", TextToSpeech.QUEUE_FLUSH, null, "ChargeFull")
                            }
                        }
                        lastBatteryLevel = level
                    }
                }
            }
        }
    }
